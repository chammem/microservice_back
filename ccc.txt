const express = require('express');
const mongoose = require('mongoose');
const { Kafka } = require('kafkajs');
const cors = require('cors');
const CommandeAnalytics = require('./models/CommandeAnalytics.model');
const OrderAnalytics = require('./models/OrderAnalytics.model');

// Configuration Express
const app = express();

// Middleware CORS (à appliquer AVANT les routes)
app.use(cors({
  origin: ['http://localhost:8100', 'http://gateway:8089'],
  methods: ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'],
  allowedHeaders: ['Content-Type', 'Authorization']
}));

app.use(express.json());

// Route API /dashboard
app.get('/api/analytics/dashboard', async (req, res) => {
  try {
    const commandes = await CommandeAnalytics.find().limit(10).sort({ createdAt: -1 });
    const orders = await OrderAnalytics.find().limit(10).sort({ createdAt: -1 });

    res.status(200).json({
      commandesCount: commandes.length,
      ordersCount: orders.length,
      commandes,
      orders
    });
  } catch (error) {
    console.error('❌ Erreur récupération dashboard:', error);
    res.status(500).json({ error: 'Erreur serveur' });
  }
});

// Connexion MongoDB
const connectDB = async () => {
  try {
    await mongoose.connect('mongodb://mongo:27017/analytics', {
      useNewUrlParser: true,
      useUnifiedTopology: true
    });
    console.log('✅ Connecté à MongoDB');
  } catch (err) {
    console.error('❌ Erreur MongoDB:', err);
    process.exit(1);
  }
};

// Configuration Kafka
const kafka = new Kafka({
  clientId: 'analytics-service',
  brokers: ['kafka:9092'],
  retry: {
    initialRetryTime: 100,
    retries: 10
  }
});

const consumer = kafka.consumer({
  groupId: 'analytics-group',
  sessionTimeout: 30000,
  heartbeatInterval: 3000,
  maxBytesPerPartition: 1048576,
  maxWaitTimeInMs: 5000
});

// Traitement des événements Kafka
async function processOrderEvent(message) {
  try {
    const order = JSON.parse(message.value.toString());
    console.log('🛒 Nouvelle commande reçue:', order.id);

    await OrderAnalytics.create({
      orderId: order.id,
      items: order.items,
      totalAmount: order.amount,
      createdAt: new Date()
    });

    console.log('📊 Statistiques mises à jour pour la commande:', order.id);
  } catch (err) {
    console.error('❌ Erreur traitement commande:', err);
  }
}

async function processCommandeEvent(message) {
  try {
    const commande = JSON.parse(message.value.toString());
    console.log('📩 Événement commande reçu:', commande.id);

    await CommandeAnalytics.create({
      commandeId: commande.id,
      client: `${commande.prenomClient} ${commande.nomClient}`.trim(),
      total: commande.totalCommande,
      statut: commande.statutCommande,
      paiement: commande.statutPaiement || 'NON_PAYE',
      createdAt: new Date()
    });

    console.log('💾 Commande enregistrée:', commande.id);
  } catch (err) {
    console.error('❌ Erreur traitement commande:', err);
  }
}

// Connexion Kafka avec reconnexion automatique
async function connectKafka() {
  const runConsumer = async () => {
    await consumer.connect();
    await consumer.subscribe({ topics: ['order-events', 'command-events'], fromBeginning: false });

    await consumer.run({
      eachMessage: async ({ topic, message }) => {
        try {
          if (topic === 'order-events') await processOrderEvent(message);
          if (topic === 'command-events') await processCommandeEvent(message);
        } catch (err) {
          console.error('❌ Erreur traitement message:', err);
        }
      }
    });
  };

  const handleError = async (err) => {
    console.error('💥 Erreur Kafka:', err);
    try {
      await consumer.disconnect();
    } catch (e) {
      console.error('Erreur déconnexion:', e);
    }
    setTimeout(connectKafka, 5000);
  };

  consumer.on('consumer.crash', handleError);
  consumer.on('consumer.disconnect', handleError);

  try {
    await runConsumer();
    console.log('👂 Consumer Kafka prêt');
  } catch (err) {
    await handleError(err);
  }
}

// Démarrage du serveur
const start = async () => {
  await connectDB();
  await connectKafka();

  const PORT = 3005; // Tu peux le changer si nécessaire
  app.listen(PORT, () => {
    console.log(`🚀 Service analytics sur le port ${PORT}`);
  });
};

// Gérer les erreurs non catchées
process.on('unhandledRejection', (err) => {
  console.error('🔥 Erreur non gérée:', err);
});

start();
